model {
  # Likelihood
  for (j in 1:3) { # 3 studies
    for (k in 1:2) { # 2 arms per study
      r[j,k] ~ dbin(p[j,k], n[j,k])
      logit(p[j,k]) <- mu[j] + delta[j,k]
    }
    # Define delta relative to arm 1
    delta[j,1] <- 0
  }
  
  # Consistency Equations (Fixed Effects)
  delta[1,2] <- d_AB
  delta[2,2] <- d_AC
  delta[3,2] <- d_AC - d_AB # This is d_BC
  
  # Priors
  for (j in 1:3) {
    # BUGS/JAGS uses Precision (1/Variance) as the 2nd parameter!
    # 0.0001 precision = 10000 variance = 100 SD (Extremely weak/vague prior)
    mu[j] ~ dnorm(0, 0.0001) 
  }
  # Vague priors for treatment effects (log odds ratios)
  d_AB ~ dnorm(0, 0.0001)
  d_AC ~ dnorm(0, 0.0001)
  
  # Derived parameters for output
  OR_AB <- exp(d_AB)
  OR_AC <- exp(d_AC)
  OR_BC <- exp(d_AC - d_AB)
}
